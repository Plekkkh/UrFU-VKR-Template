\chapter{Расчетная часть}

\section{Расчет воздухообмена для разбавления вредных веществ}
По формуле~\eqref{eq:gas2} для каждого вещества ($q_{\text{уд.}} - q_{\text{пр.}} = 0{,}7\,\text{ПДК}$):

Кислота муравьиная:
\begin{equation*}
    L_{1} = \frac{3 \cdot 10^{4}}{0{,}7 \cdot 1{,}0} = 42\,857{,}1~\text{м}^3/\text{ч}.
\end{equation*}

Марганец:
\begin{equation*}
    L_{2} = \frac{7 \cdot 10^{2}}{0{,}7 \cdot 0{,}2} = 5\,000{,}0~\text{м}^3/\text{ч}.
\end{equation*}

Этанол:
\begin{equation*}
    L_{3} = \frac{3 \cdot 10^{6}}{0{,}7 \cdot 1000} = 4\,285{,}7~\text{м}^3/\text{ч}.
\end{equation*}

Фенол:
\begin{equation*}
    L_{4} = \frac{4 \cdot 10^{2}}{0{,}7 \cdot 0{,}3} = 1\,904{,}8~\text{м}^3/\text{ч}.
\end{equation*}

Никель:
\begin{equation*}
    L_{5} = \frac{3 \cdot 10^{2}}{0{,}7 \cdot 0{,}05} = 8\,571{,}4~\text{м}^3/\text{ч}.
\end{equation*}

\section{Расчет воздухообмена для удаления избыточного тепла}
Температура удаляемого воздуха по формуле~\eqref{eq:tud}:
\begin{equation*}
    t_{\text{уд.}} = 26 + 1 \cdot (5{,}5 - 2) = 29{,}5~^{\circ}\text{C}.
\end{equation*}

Количество приточного воздуха по формуле~\eqref{eq:heat}:
\begin{equation*}
    L_{6} = \frac{5 \cdot 10^{3}}{0{,}241 \cdot 1{,}130 \cdot (29{,}5 - 26)} = \frac{5 \cdot 10^{3}}{0{,}953} \approx 5\,245{,}7~\text{м}^3/\text{ч}.
\end{equation*}

\section{Определение расчетного воздухообмена и кратности}
При одновременном выделении вредных веществ и теплоты за расчетный принимается наибольшее из полученных значений:
\begin{equation*}
    L_{\text{пр.}} = \max(L_{1}, \ldots, L_{6}) = 42\,857{,}1~\text{м}^3/\text{ч}.
\end{equation*}

Кратность воздухообмена по формуле~\eqref{eq:kr}:
\begin{equation*}
    K_{\text{р}} = \frac{42\,857{,}1}{7200} = 5{,}95~\text{ч}^{-1}.
\end{equation*}
