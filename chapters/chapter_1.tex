\section{Расчет расстояний и вспомогательных коэффициентов}

Расстояние от смартфона до базовых станций определяется по формуле Евклидовой метрики:
\begin{equation}
    d = \sqrt{(X_{\text{см}} - X_{\text{БС}})^2 + (Y_{\text{см}} - Y_{\text{БС}})^2}.
    \label{eq:distance}
\end{equation}

Подставляя численные значения координат, находим расстояния до БС №1 ($d_1$), БС №2 ($d_2$) и ложной БС ($d_{\text{Fake}}$):
\begin{align*}
    d_1 &= \sqrt{(390 - 200)^2 + (290 - 380)^2} = \sqrt{190^2 + (-90)^2} \\
        &= \sqrt{36100 + 8100} = \sqrt{44200} \approx 210.24\text{ м} = 0.21024\text{ км},
\end{align*}
\begin{align*}
    d_2 &= \sqrt{(390 - 850)^2 + (290 - 420)^2} = \sqrt{(-460)^2 + (-130)^2} \\
        &= \sqrt{211600 + 16900} = \sqrt{228500} \approx 478.02\text{ м} = 0.47802\text{ км},
\end{align*}
\begin{align*}
    d_{\text{Fake}} &= \sqrt{(390 - 480)^2 + (290 - 240)^2} = \sqrt{(-90)^2 + (50)^2} \\
                    &= \sqrt{8100 + 2500} = \sqrt{10600} \approx 102.96\text{ м} = 0.10296\text{ км}.
\end{align*}

Функция коррекции высоты мобильной станции для средних и крупных городов рассчитывается по формуле:
\begin{equation}
    a(h_{\text{см}}) = 3.2\log_{10}(11.75 \cdot h_{\text{см}}) - 4.97.
    \label{eq:height_correction}
\end{equation}
Подставляя $h_{\text{см}} = 1.5$ м, получаем:
\begin{align*}
    a(1.5) &= 3.2\log_{10}(11.75 \cdot 1.5) - 4.97 = 3.2\log_{10}(17.625) - 4.97 \\
           &\approx 3.2 \cdot 1.24613 - 4.97 = -0.982384 \approx -0.9824\text{ дБ}.
    \label{eq:height_correction_value}
\end{align*}

Коэффициент наклона трассы спада мощности определяется выражением:
\begin{equation}
    n_{\text{slope}} = 44.9 - 6.55\log_{10}(h_{\text{БС}}).
    \label{eq:slope_coefficient}
\end{equation}
При высоте подвеса антенны базовой станции $h_{\text{БС}} = 30\text{ м}$:
\begin{equation*}
    n_{\text{slope}} = 44.9 - 6.55\log_{10}(30) \approx 44.9 - 6.55 \cdot 1.47712 \approx 35.225\text{ дБ/дек}.
\end{equation*}

\section{Расчет уровней пилот-сигналов RSRP в режиме LTE}

Потери радиосигнала на эталонном расстоянии 1 км по модели COST 231 Hata:
\begin{equation}
    L_{1\text{км}} = 46.3 + 33.9\log_{10}(f) - 13.82\log_{10}(h_{\text{БС}}) - a(h_{\text{см}}) + C_m.
    \label{eq:loss_1km}
\end{equation}

Полные трассовые потери радиосигнала в городской застройке на расстоянии $d_{\text{км}}$:
\begin{equation}
    L_{\text{TR}} = L_{1\text{км}} + n_{\text{slope}}\log_{10}(d_{\text{км}}).
    \label{eq:total_loss}
\end{equation}

Уровень принимаемого пилот-сигнала $\mathrm{RSRP}$ на антенном входе смартфона:
\begin{equation}
    \mathrm{RSRP} = P_{\text{TX}} - L_{\text{TR}}.
    \label{eq:rsrp}
\end{equation}

Выполним расчет для БС №1 LTE с данными из таблицы~\ref{tab:input_data}: $f_{\text{БС1 LTE}} = 2620\text{ МГц}$, $P_{\text{TX1}} = 18\text{ дБм}$, $d_1 = 0.21024\text{ км}$:
\begin{align*}
    L_{1\text{км (БС1 LTE)}} &= 46.3 + 33.9\log_{10}(2620) - 13.82\log_{10}(30) - (-0.9824) + 3.0 \\
                             &\approx 145.75\text{ дБ},
\end{align*}
\begin{equation*}
    L_{\text{TR (БС1 LTE)}} = 145.75 + 35.225\log_{10}(0.21024) \approx 145.75 - 23.86 = 121.89\text{ дБ},
\end{equation*}
\begin{equation*}
    \mathrm{RSRP}_{\text{БС1}} = 18 - 121.89 = -103.89\text{ дБм}.
\end{equation*}

Выполним расчет для БС №2 LTE с данными: $f_{\text{БС2 LTE}} = 2670\text{ МГц}$, $P_{\text{TX2}} = 25\text{ дБм}$, $d_2 = 0.47802\text{ км}$:
\begin{align*}
    L_{1\text{км (БС2 LTE)}} &= 46.3 + 33.9\log_{10}(2670) - 13.82\log_{10}(30) - (-0.9824) + 3.0 \\
                             &\approx 146.03\text{ дБ},
\end{align*}
\begin{equation*}
    L_{\text{TR (БС2 LTE)}} = 146.03 + 35.225\log_{10}(0.47802) \approx 146.03 - 11.29 = 134.74\text{ дБ},
\end{equation*}
\begin{equation*}
    \mathrm{RSRP}_{\text{БС2}} = 25 - 134.74 = -109.74\text{ дБм}.
\end{equation*}

\section{Расчет уровня сигнала $\mathrm{RxLev}$ и критериев C1 в режиме GSM}

Для диапазона GSM-900 применяется модель Окамура-Хата, в которой потери на расстоянии 1 км определяются уравнением:
\begin{equation}
    L_{1\text{км}} = 69.55 + 26.16\log_{10}(f) - 13.82\log_{10}(h_{\text{БС}}) - a(h_{\text{см}}).
    \label{eq:loss_1km_gsm}
\end{equation}

Уровень принимаемого сигнала $\mathrm{RxLev}$ и критерий выбора соты $C1$ рассчитывается по формулам:
\begin{align}
    \mathrm{RxLev} &= P_{\text{TX}} - L_{\text{TR}}, \label{eq:rxlev} \\
    C1 &= \mathrm{RxLev} - \mathrm{AccMin}. \label{eq:c1}
\end{align}

Выполним расчет для БС №1 GSM с данными: $f_{\text{БС1 GSM}} = 935\text{ МГц}$, $P_{\text{TX1}} = 18\text{ дБм}$, $\mathrm{AccMin} = -115.0\text{ дБм}$:
\begin{align*}
    L_{1\text{км (БС1 GSM)}} &= 69.55 + 26.16\log_{10}(935) - 13.82\log_{10}(30) - (-0.9824) \\
                             &\approx 127.83\text{ дБ},
\end{align*}
\begin{equation*}
    L_{\text{TR (БС1 GSM)}} = 127.83 + 35.225\log_{10}(0.21024) \approx 127.83 - 23.86 = 103.97\text{ дБ},
\end{equation*}
\begin{equation*}
    \mathrm{RxLev}_{\text{БС1}} = 18 - 103.97 = -85.97\text{ дБм},
\end{equation*}
\begin{equation*}
    C1_{\text{БС1}} = -85.97 - (-115.0) = 29.03\text{ дБ}.
\end{equation*}

Выполним расчет для БС №2 GSM с данными: $f_{\text{БС2 GSM}} = 955\text{ МГц}$, $P_{\text{TX2}} = 25\text{ дБм}$, $\mathrm{AccMin} = -115.0\text{ дБм}$:
\begin{align*}
    L_{1\text{км (БС2 GSM)}} &= 69.55 + 26.16\log_{10}(955) - 13.82\log_{10}(30) - (-0.9824) \\
                             &\approx 128.08\text{ дБ},
\end{align*}
\begin{equation*}
    L_{\text{TR (БС2 GSM)}} = 128.08 + 35.225\log_{10}(0.47802) \approx 128.08 - 11.29 = 116.79\text{ дБ},
\end{equation*}
\begin{equation*}
    \mathrm{RxLev}_{\text{БС2}} = 25 - 116.79 = -91.79\text{ дБм},
\end{equation*}
\begin{equation*}
    C1_{\text{БС2}} = -91.79 - (-115.0) = 23.21\text{ дБ}.
\end{equation*}

\section{Расчет мощности ложной БС и максимального расстояния перехвата}

Рассчитаем трассовые потери для сигнала Fake БС с данными: $f_{\text{Fake GSM}} = 945\text{ МГц}$, $d_{\text{Fake}} = 0.10296\text{ км}$:
\begin{align*}
    L_{1\text{км (Fake)}} &= 69.55 + 26.16\log_{10}(945) - 13.82\log_{10}(30) - (-0.9824) \approx 127.96\text{ дБ}, \\
    L_{\text{TR (Fake)}} &= 127.96 + 35.225\log_{10}(0.10296) \approx 127.96 - 34.78 = 93.18\text{ дБ}.
\end{align*}

Для перехвата трафика критерий выбора ложной соты должен превышать максимальный критерий легитимной сети:
\begin{equation}
    P_{\text{Fake}} > C1_{\text{БС1}} + L_{\text{TR (Fake)}} + \mathrm{AccMin}_{\text{Fake БС}}.
    \label{eq:fake_power}
\end{equation}
При $\mathrm{AccMin_{\text{Fake БС}}} = -100.0\text{ дБм}$, получаем:
\begin{equation*}
    P_{\text{Fake}} > 29.03 + 93.18 + (-100.0) = 22.21\text{ дБм}.
\end{equation*}

Максимальное расстояние до ложной БС при конспиративной мощности $P_{\text{Fake}} = P_{\text{TX1}} = 18\text{ дБм}$ определяется из уравнения:
\begin{equation}
    d_{\max} \approx 10^{\dfrac{P_{\text{Fake}} - \mathrm{RxLev}_{\text{БС1}} - L_{1\text{км (Fake)}}}{n_{\text{slope}}}}.
    \label{eq:max_distance}
\end{equation}
\begin{equation*}
    d_{\max} \approx 10^{\frac{18 - (-85.97) - 127.96}{35.225}} \approx 10^{-0.68105} \approx 0.2084\text{ км} = 208.4\text{ м}.
\end{equation*}

С учетом заданного порога $\mathrm{AccMin}_{\text{Fake БС}} = -100.0\text{ дБм}$, требуемый уровень сигнала от ложной БС для достижения $C1_{\text{Fake}} = C1_{\text{БС1}} = 29.03\text{ дБ}$ составляет $\mathrm{RxLev}_{\text{Fake треб}} = C1_{\text{БС1}} + \mathrm{AccMin}_{\text{Fake БС}} = 29.03 + (-100.0) = -70.97\text{ дБм}$.
Тогда при мощности $P_{\text{Fake}} = 18\text{ дБм}$, максимальное расстояние перехвата равно:
\begin{equation*}
    d_{\max (-100\text{ дБм})} \approx 10^{\frac{18 - (-70.97) - 127.96}{35.225}} \approx 10^{-1.10688} \approx 0.07818\text{ км} = 78.18\text{ м}.
\end{equation*}
