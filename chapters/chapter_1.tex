\section{Расчет АМ-сигнала}\label{sec:AM_calculation}

\textbf{Дано:} $U_0 = 1.0$ В, $m = 0.8$, частота сообщения $F = 1$ кГц, частота несущей $f_0 = 12.0$ кГц.

\begin{enumerate}
    \item Расчет максимальной и минимальной амплитуды:
    \begin{align*}
        U_{\max} &= U_0 (1 + m) = 1.0 \cdot (1 + 0.8) = 1.8 \text{ В;} \\
        U_{\min} &= U_0 (1 - m) = 1.0 \cdot (1 - 0.8) = 0.2 \text{ В.}
    \end{align*}
    \item Расчет коэффициента модуляции:
    \begin{equation}
        m = \frac{U_{\max} - U_{\min}}{U_{\max} + U_{\min}}
        \label{eq:AM_modulation_index}
    \end{equation}
    Таким образом:
    \begin{equation*}
        m = \frac{1.8 - 0.2}{1.8 + 0.2} = \frac{1.6}{2.0} = 0.8.
    \end{equation*}
    \item Расчет ширины полосы частот:
    \begin{equation*}
        \Delta f = 2 \cdot F = 2 \cdot 1 \text{ кГц} = 2 \text{ кГц.}
    \end{equation*}
    \item Расчет частот и уровней спектральных составляющих:
    \begin{itemize}
        \item Центральная несущая: $f_0 = 12.0$ кГц, амплитуда $U_{\text{нес}} = U_0 = 1.0$ В;
        \item Нижняя боковая: $f_{\text{нб}} = f_0 - F = 12.0 - 1.0 = 11.0$ кГц;
        \item Верхняя боковая: $f_{\text{вб}} = f_0 + F = 12.0 + 1.0 = 13.0$ кГц;
        \item Амплитуда боковых:
        \begin{equation*}
            U_{\text{нб}} = U_{\text{вб}} = \frac{1}{2} \cdot m \cdot U_0 = \frac{1}{2} \cdot 0.8 \cdot 1.0 = 0.4 \text{ В.}
        \end{equation*}
    \end{itemize}
    \item Длина линии фазового портрета на комплексной IQ-плоскости:
    \begin{equation*}
        L = 2 \cdot m \cdot U_0 = 2 \cdot 0.8 \cdot 1.0 = 1.6 \text{ В.}
    \end{equation*}
\end{enumerate}

\section{Расчет ЧМ-сигнала}

\textbf{Дано:} $U_0 = 1.0$ В, $F = 1$ кГц, $f_0 = 12.0$ кГц, $m = 0.2; 5.0$


Расчет частот спектральных составляющих:
\begin{itemize}
    \item Центральная несущая: $f_0 = 12.0$ кГц;
    \item Нижние боковые:
    \begin{align*}
        f_{-1} &= f_0 - F = 12.0 - 1.0 = 11.0 \text{ кГц;} \\
        f_{-2} &= f_0 - 2F = 12.0 - 2 \cdot 1.0 = 10.0 \text{ кГц;}
    \end{align*}
    \item Верхние боковые:
    \begin{align*}
        f_{+1} &= f_0 + F = 12.0 + 1.0 = 13.0 \text{ кГц;} \\
        f_{+2} &= f_0 + 2F = 12.0 + 2 \cdot 1.0 = 14.0 \text{ кГц.}
    \end{align*}
\end{itemize}

Расчет для $ m = 0.2 $:
\begin{enumerate}
    \item Девиация частоты:
    \begin{equation*}
        \Delta f_M = m \cdot F = 0.2 \cdot 1.0 = 0.2 \text{ кГц.}
    \end{equation*}
    \item Ширина полосы частот:
    \begin{equation*}
        \Delta f_c = 2 \cdot F = 2 \cdot 1.0 = 2.0 \text{ кГц.}
    \end{equation*}
    \item Амплитуды спектральных составляющих:
    \begin{itemize}
        \item Центральная несущая ($f_0 = 12.0$ кГц): $A_0 = U_0 = 1.0$ В;
        \item Первая пара боковых ($f_{-1} = 11.0$ кГц, $f_{+1} = 13.0$ кГц): $A_1 = \frac{1}{2} \cdot m \cdot U_0 = \frac{1}{2} \cdot 0.2 \cdot 1.0 = 0.1$ В;
        \item Вторая пара боковых ($f_{-2} = 10.0$ кГц, $f_{+2} = 14.0$ кГц): $A_2 \approx 0$ В (при малом $m$ пренебрежимо малы).
    \end{itemize}
\end{enumerate}

Расчет для $ m = 5.0 $:
\begin{enumerate}
    \item Девиация частоты:
    \begin{equation*}
        \Delta f_M = m \cdot F = 5.0 \cdot 1.0 = 5.0 \text{ кГц.}
    \end{equation*}
    \item Ширина полосы частот:
    \begin{equation*}
        \Delta f_c = 2 \cdot F \cdot (m + 1) = 2 \cdot 1.0 \cdot (5.0 + 1.0) = 12.0 \text{ кГц.}
    \end{equation*}
    \item Амплитуды спектральных составляющих:
    \begin{itemize}
        \item Центральная несущая ($f_0 = 12.0$ кГц): $A_0 = U_0 \cdot |J_0(5)| = 1.0 \cdot |-0.1776| = 0.178$ В;
        \item Первая пара боковых ($f_{-1} = 11.0$ кГц, $f_{+1} = 13.0$ кГц): $A_1 = U_0 \cdot |J_1(5)| = 1.0 \cdot |-0.3276| = 0.328$ В;
        \item Вторая пара боковых ($f_{-2} = 10.0$ кГц, $f_{+2} = 14.0$ кГц): $A_2 = U_0 \cdot |J_2(5)| = 1.0 \cdot |0.0466| = 0.047$ В.
    \end{itemize}
\end{enumerate}

\section{Расчет информационной скорости}

Информационная скорость рассчитывается по формуле связи с символьной скоростью $R_s$:
\begin{equation*}
    R = R_s \cdot n = R_s \cdot \log_2(M)
\end{equation*}

\begin{itemize}
    \item ФМ-2: $M = 2 \Rightarrow R = R_s \cdot \log_2(2) = R_s \cdot 1$;
    \item ФМ-4: $M = 4 \Rightarrow R = R_s \cdot \log_2(4) = R_s \cdot 2$;
    \item APSK-16: $M = 16 \Rightarrow R = R_s \cdot \log_2(16) = R_s \cdot 4$;
    \item QAM-32: $M = 32 \Rightarrow R = R_s \cdot \log_2(32) = R_s \cdot 5$;
\end{itemize}

\textit{Примечание:} в виртуальном стенде $R_s = 1$ кБод, поэтому скорости равны соответственно 1, 2, 4 и 5 кбит/с.

\section{Расчет пиковой амплитуды $U_m$ для QAM-16 и QAM-32}
\label{sec:qam-16--qam-32_measurements}

Формула расстояния от центра до наиболее удаленной точки:
\begin{equation*}
    U_m = \sqrt{I_{\max}^2 + Q_{\max}^2}
\end{equation*}

\begin{itemize}
    \item QAM-16:\\
    Сетка амплитуд: $I,Q \in [-3, -1, +1, +3] \cdot \frac{U_0}{3}$.\\
    Максимальные координаты угловой точки: $I_{\max} = 3 \cdot \frac{1.0}{3} = 1.0$ В;
    $Q_{\max} = 3 \cdot \frac{1.0}{3} = 1.0$ В.\\
    Пиковая амплитуда:
    \begin{equation*}
        U_m = \sqrt{1.0^2 + 1.0^2} = \sqrt{2} \approx 1.414 \text{ В.}
    \end{equation*}
    \item QAM-32:\\
    Сетка амплитуд: $I,Q \in [-5, -3, -1, +1, +3, +5] \cdot \frac{U_0}{4.5}$.\\
    Так как в схеме QAM-32 4 крайних угла созвездия вырезаны, то наиболее удаленными являются точки с координатами $(\pm 5, \pm 3)$ или $(\pm 3, \pm 5)$: \\
    $I = 5 \cdot \frac{1.0}{4.5} = 1.111$ В;
    $Q = 3 \cdot \frac{1.0}{4.5} = 0.667$ В.\\
    Пиковая амплитуда:
    \begin{equation*}
        U_m = \sqrt{(\frac{5}{4.5})^2 + (\frac{3}{4.5})^2} = \frac{\sqrt{25 + 9}}{4.5} = \frac{\sqrt{34}}{4.5} \approx \frac{5.831}{4.5} \approx 1.296 \text{ В.}
    \end{equation*}
    \textit{Примечание:} Если пренебречь вырезанием углов и брать полный квадрат 6 на 6, то пиковая амплитуда будет равна $1.571$ В.
\end{itemize}
