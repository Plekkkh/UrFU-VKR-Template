\section{Расчет уровня собственного теплового шума приемника}

Уровень собственного теплового шума $N$ приемника в полосе пропускания $\mathrm{BW} = 100\text{ кГц} = 10^5\text{ Гц}$ при коэффициенте шума $\mathrm{NF} = 6\text{ дБ}$ определяется выражением:
\begin{equation}
    N = -174 + 10 \cdot \lg(\mathrm{BW}) + \mathrm{NF}
    \label{eq:noise_level}
\end{equation}
Подставляя численные значения, получим:
\begin{equation*}
    N = -174 + 10 \cdot \lg(10^5) + 6 = -174 + 50 + 6 = -118.0\text{ дБм}.
\end{equation*}

\section{Расчет затухания ПЭМИ на трассе распространения}
Для частоты излучения $f = 148.5\text{ МГц}$ (диапазон $47.75\text{ МГц} < f \leqslant 1800\text{ МГц}$) коэффициент затухания $L_V$ (в разах по напряжению) рассчитывается по формуле:
\begin{equation}
    L_V \approx
    \begin{cases}
        R^2, & \text{если } R \leqslant \dfrac{1800}{f}; \\[14pt]
        \dfrac{1800 \cdot R}{f}, & \text{если } R > \dfrac{1800}{f}.
    \end{cases}
    \label{eq:attenuation_coefficient}
\end{equation}

Определим граничное расстояние перехода между зонами затухания:
\begin{equation}
    r_{\text{гр}} = \frac{1800}{f} = \frac{1800}{148.5} = 12.12\text{ м}.
    \label{eq:boundary_distance}
\end{equation}

Перевод коэффициента затухания в децибелы осуществляется по формуле:
\begin{equation}
    L_V\text{ [дБ]} = 20 \cdot \lg(L_V).
    \label{eq:attenuation_coefficient_db}
\end{equation}

Выполним расчет для заданных дистанций.
При $R = 1\text{ м}$:
\begin{equation*}
    L_V(1) = 1^2 = 1\text{ раз} \Rightarrow L_V\text{ [дБ]} = 20\lg(1) = 0.00\text{ дБ}.
\end{equation*}

При $R = 2\text{ м}$:
\begin{equation*}
    L_V(2) = 2^2 = 4\text{ раза} \Rightarrow L_V\text{ [дБ]} = 20\lg(4) = 12.04\text{ дБ}.
\end{equation*}

При $R = 5\text{ м}$:
\begin{equation*}
    L_V(5) = 5^2 = 25\text{ раз} \Rightarrow L_V\text{ [дБ]} = 20\lg(25) = 27.96\text{ дБ}.
\end{equation*}

При $R = 10\text{ м}$:
\begin{equation*}
    L_V(10) = 10^2 = 100\text{ раз} \Rightarrow L_V\text{ [дБ]} = 20\lg(100) = 40.00\text{ дБ}.
\end{equation*}

При $R = 20\text{ м}$:
\begin{equation*}
    L_V(20) = \dfrac{1800 \cdot 20}{148.5} = 242.42\text{ раза} \Rightarrow L_V\text{ [дБ]} = 20\lg(242.42) = 47.69\text{ дБ}.
\end{equation*}

При $R = 30\text{ м}$:
\begin{equation*}
    L_V(30) = \dfrac{1800 \cdot 30}{148.5} = 363.64\text{ раза} \Rightarrow L_V\text{ [дБ]} = 20\lg(363.64) = 51.21\text{ дБ}.
\end{equation*}

При $R = 50\text{ м}$:
\begin{equation*}
    L_V(50) = \dfrac{1800 \cdot 50}{148.5} = 606.06\text{ раза} \Rightarrow L_V\text{ [дБ]} = 20\lg(606.06) = 55.65\text{ дБ}.
\end{equation*}

При $R = 80\text{ м}$:
\begin{equation*}
    L_V(80) = \dfrac{1800 \cdot 80}{148.5} = 969.70\text{ раза} \Rightarrow L_V\text{ [дБ]} = 20\lg(969.70) = 59.73\text{ дБ}.
\end{equation*}

При $R = 100\text{ м}$:
\begin{equation*}
    L_V(100) = \dfrac{1800 \cdot 100}{148.5} = 1212.12\text{ раза} \Rightarrow L_V\text{ [дБ]} = 20\lg(1212.12) = 61.67\text{ дБ}.
\end{equation*}

\section{Расчет мощности сигнала на входе приемника и отношения сигнал/шум}

Мощность сигнала $P_{rx}$ на входе приемника и отношение сигнал/шум $\mathrm{SNR}$ определяются выражениями:
\begin{equation}
    P_{rx} = P_{tx} + G_{rx} - L_V\text{ [дБ]},
    \label{eq:received_power}
\end{equation}
\begin{equation}
    \mathrm{SNR} = P_{rx} - N.
    \label{eq:SNR}
\end{equation}

Пример расчета для дистанции $R = 2\text{ м}$ ($L_V = 12.04\text{ дБ}$):

Всенаправленная антенна ($G_{rx1} = 2.15\text{ дБи}$):
\begin{equation*}
    P_{rx1}(2) = -68.0 + 2.15 - 12.04 = -77.89\text{ дБм},
\end{equation*}
\begin{equation*}
    \mathrm{SNR}_1(2) = -77.89 - (-118.0) = +40.11\text{ дБ}.
\end{equation*}

Направленная антенна ($G_{rx2} = 12.0\text{ дБи}$):
\begin{equation*}
    P_{rx2}(2) = -68.0 + 12.0 - 12.04 = -68.04\text{ дБм},
\end{equation*}
\begin{equation*}
    \mathrm{SNR}_2(2) = -68.04 - (-118.0) = +49.96\text{ дБ}.
\end{equation*}

Определим максимальную расчетную дальность обнаружения сигнала из условия $\mathrm{SNR} = 0\text{ дБ}$ (предельное затухание $L_{V\max} = P_{tx} + G_{rx} - N$):

Для всенаправленной антенны: $L_{V\max1} = -68 + 2.15 - (-118) = 52.15\text{ дБ}$ ($405.0\text{ раз}$), откуда:
\begin{equation*}
    R_{\max1} = \frac{L_{V\max1} \cdot f}{1800} = \frac{405.0 \cdot 148.5}{1800} = 33.4\text{ м}.
\end{equation*}

Для направленной антенны: $L_{V\max2} = -68 + 12.0 - (-118) = 62.00\text{ дБ}$ ($1258.9\text{ раза}$), откуда:
\begin{equation*}
    R_{\max2} = \frac{L_{V\max2} \cdot f}{1800} = \frac{1258.9 \cdot 148.5}{1800} = 103.9\text{ м}.
\end{equation*}

Получается, что замена всенаправленной антенны на направленную дает выигрыш по мощности $\Delta G = 12.0 - 2.15 = 9.85\text{ дБ}$ и увеличивает расчетную дальность перехвата в дальней зоне в $10^{9.85/20} = 3.11\text{ раза}$.

\section{Расчет снижения уровня шума при сужении полосы ПЧ вольтметра}

При сужении полосы ПЧ селективного микровольтметра с $\Delta f_1 = 200\text{ кГц}$ до $\Delta f_2 = 0.2\text{ кГц}$ мощность шума снижается пропорционально отношению полос, а действующее напряжение шума $U_{\text{ш}}$ на табло вольтметра — пропорционально квадратному корню из отношения полос:
\begin{equation}
    K_U = \sqrt{\frac{\Delta f_1}{\Delta f_2}} = \sqrt{\frac{200\text{ кГц}}{0.2\text{ кГц}}} = \sqrt{1000} = 31.62\text{ раза}.
    \label{eq:noise_reduction_factor}
\end{equation}

\section{Расчет параметров структуры полного видеосигнала $\mathbf{1920 \times 1080}$ ($\mathbf{60}\text{ Гц}$)}

Количество активных строк в кадре: $N_{\text{акт}} = h = 1080\text{ строк}$.
Общее количество строк в кадре (активные + служебные):
\begin{equation}
    N_{\text{полн}} = h + V_{porch} = 1080 + 45 = 1125\text{ строк}.
    \label{eq:total_lines}
\end{equation}

Длительность строчного гасящего интервала во времени:
\begin{equation}
    t_{H\_porch} = \frac{H_{porch}}{f_{pixel}} = \frac{280}{148.5 \cdot 10^6\text{ Гц}} = 1.89 \cdot 10^{-6}\text{ с} = 1.89\text{ мкс}.
    \label{eq:horizontal_porch}
\end{equation}

Длительность кадрового гасящего интервала во времени:
\begin{equation}
    t_{V\_porch} = \frac{V_{porch}}{f_H} = \frac{45}{67.5 \cdot 10^3\text{ Гц}} = 0.667 \cdot 10^{-3}\text{ с} = 0.667\text{ мс}.
    \label{eq:vertical_porch}
\end{equation}
