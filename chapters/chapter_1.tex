\chapter{Расчеты домашнего задания}\label{ch:homework}

\section{Методика расчета}

Интегральный индекс артикуляции $R$ определяется по формуле:
\begin{equation}
    R = \sum_{i=2}^{6} k_i P_i,
    \label{eq:equation}
\end{equation}
где $k_i$ --- весовой коэффициент $i$-й октавной полосы, $P_i$ --- коэффициент восприятия формант в $i$-й полосе частот.

Функция формантного распределения $k(f)$ аппроксимируется выражением:
\begin{equation}
    k(f) =
    \begin{cases}
        2{,}57 \cdot 10^{-8} \cdot f^{2{,}4}, & \text{при } 100 < f \leqslant 400~\text{Гц}, \\
        1 - 1{,}074 \exp\left(-10^{-4} \cdot f^{1{,}18}\right), & \text{при } 400 < f < 10000~\text{Гц}.
    \end{cases}
    \label{eq:equation2}
\end{equation}
Весовой коэффициент полосы определяется как разность значений функции на границах октавы:
\begin{equation}
    k_i = k(f_{\text{в}i}) - k(f_{\text{н}i}).
    \label{eq:equation3}
\end{equation}

Пример расчета весового коэффициента для полосы 2 ($f_{\text{н}2} = 1800~\text{Гц}$, $f_{\text{в}2} = 355~\text{Гц}$):
\begin{equation*}
    k_2 = 2{,}57 \cdot 10^{-8} \cdot 355^{2{,}4} - 2{,}57 \cdot 10^{-8} \cdot 180^{2{,}4} = 0{,}03390 - 0{,}00662 = 0{,}02728.
\end{equation*}

Формантный параметр $\Delta A_i$, характеризующий энергетическую избыточность дискретных составляющих русской речи, вычисляется по формуле:
\begin{equation}
    \Delta A_i =
    \begin{cases}
        200 \cdot f_i^{-0{,}43} - 0{,}37, & \text{при } f_i \leqslant 1000~\text{Гц}, \\
        1{,}37 + \dfrac{1000}{f_i^{0{,}69}}, & \text{при } f_i > 1000~\text{Гц}.
    \end{cases}
    \label{eq:equation4}
\end{equation}

Пример расчета формантного параметра для полосы 2 ($f_2 = 250~\text{Гц}$):
\begin{equation*}
    \Delta A_2 = 200 \cdot 250^{-0{,}43} - 0{,}37 = 18{,}6173 - 0{,}37 = 18{,}2473~\text{дБ}.
\end{equation*}

Отношение сигнал/шум $q_i$ за ограждающей конструкцией и относительный уровень интенсивности формант $Q_i$ определяются по формулам:
\begin{equation}
    q_i = L_{\text{с}i} - L_{\text{ш}i} - a_{\text{огр }i},
    \label{eq:equation5}
\end{equation}
\begin{equation}
    Q_i = q_i - \Delta A_i.
    \label{eq:equation6}
\end{equation}

Пример расчета для акустического канала и перегородки из стекла в полосе 2 ($L_{\text{с}2} = 72~\text{дБ}$, $L_{\text{ш}2} = 25{,}2~\text{дБ}$, $a_{\text{огр }2} = 14{,}2~\text{дБ}$):
\begin{equation*}
    q_2 = 72 - 25{,}2 - 14{,}2 = 32{,}6~\text{дБ}, \quad Q_2 = 32{,}6 - 18{,}2473 = 14{,}3527~\text{дБ}.
\end{equation*}

Коэффициент восприятия формант $P(Q_i)$ определяется выражением:
\begin{equation}
    P(Q_i) =
    \begin{cases}
        \dfrac{0{,}78 + 5{,}46 \exp\left(-4{,}3 \cdot 10^{-3} (27{,}3 - |Q_i|)^2\right)}{1 + 10^{0{,}1 |Q_i|}}, & \text{при } Q_i \leqslant 0, \\[14pt]
        1 - \dfrac{0{,}78 + 5{,}46 \exp\left(-4{,}3 \cdot 10^{-3} (27{,}3 - |Q_i|)^2\right)}{1 + 10^{0{,}1 |Q_i|}}, & \text{при } Q_i > 0.
    \end{cases}
    \label{eq:equation7}
\end{equation}

Пример расчета $P_2$ при $Q_2 = 14{,}3527~\text{дБ}$:
\begin{equation*}
    P_2 = 1 - \frac{0{,}78 + 5{,}46 \exp\left(-0{,}0043 \cdot (27{,}3 - 14{,}3527)^2\right)}{1 + 10^{0{,}1 \cdot 14{,}3527}} = 1 - \frac{3{,}4355}{28{,}2442} = 0{,}8784.
\end{equation*}


Словесная разборчивость речи $W$ рассчитывается по следующему соотношению:
\begin{equation}
    W =
    \begin{cases}
        1{,}54 \cdot R^{0{,}25} \left(1 - \exp(-11 R)\right), & \text{при } R \leqslant 0{,}15, \\[6pt]
        1 - \exp\left(-\dfrac{11 R}{1 + 0{,}7 R}\right), & \text{при } R > 0{,}15.
    \end{cases}
    \label{eq:equation8}
\end{equation}

Пример расчета $W$ для перегородки из стекла при акустическом канале ($R = 0{,}7213$):
\begin{equation*}
    W = \left(1 - \exp\left(-\frac{11 \cdot 0{,}7213}{1 + 0{,}7 \cdot 0{,}7213}\right)\right) \cdot 100\,\% = 99{,}49\,\%.
\end{equation*}
